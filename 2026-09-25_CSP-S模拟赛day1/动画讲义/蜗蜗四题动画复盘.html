翻转只影响两个数都在区间里的那些对：顺序对全变逆序对，逆序对全变顺序对。
所以：翻转 [l, r] 的收益 = 区间里的顺序对 − 区间里的逆序对。

区间长 len，里面一共 len×(len−1)/2 对，每对不是顺序就是逆序（排列里没有相等的数）。

所以 顺序对 = len×(len−1)/2 − 逆序对。

收益 = 顺序对 − 逆序对 = len×(len−1)/2 − 2 × 区间逆序对。

答案 = 原来的逆序对 + max(0, 所有区间里最大的收益)。

第二步　数据范围允许多慢？
T ≤ 10⁵，n ≤ 8000，看起来很吓人。但最后一句是 ∑n ≤ 8000：所有数据的 n 加起来不超过 8000。一组 n = 8000 的数据，和 8000 组 n = 1 的数据，是同一个总量。

每组 O(n²) 时，所有组加起来 ≤ (∑n)² = 8000² = 6.4×10⁷，1.5 秒随便过。

区间一共约 n²/2 = 3.2×10⁷ 个。只要每个区间的逆序对能 O(1) 得到，就能全部枚举。

第三步（核心）　怎么 O(1) 得到每个区间的逆序对
直接对每个区间重新数一遍逆序对，一个区间就要 O(n²)，太慢。那就想：相邻的两个区间差在哪？
[l, r] 比 [l, r−1] 只多了最后一个数 a[r]。多出来的逆序对，只可能是“a[r] 和它前面的数”组成的对，也就是 [l, r−1] 里比 a[r] 大的那些数。

inv(l, r) = inv(l, r−1) + （[l, r−1] 里比 a[r] 大的个数）

在表格里就是：这一格 = 左边那一格 + 一个数。

剩下的问题：“[l, r−1] 里比 a[r] 大的个数”怎么 O(1) 拿到？
固定 r，让 l 从 r−1 开始往左走。每走一步，[l, r−1] 就多包进来一个数 a[l]，而之前包进来的数一个都不会离开。所以只要看新来的 a[l] 是不是比 a[r] 大，是就 cnt + 1。cnt 从头到尾只会累加，每步 O(1)。

inline void solve () {

#include <bits/stdc++.h>
using namespace std;
#define int long long
int T;
int a[1000010], cur[1000010];
vector <int> ops;
inline void solve () {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    for (int i = 1; i <= n; i++) {
        if (a[i] > i) {
            cout << "NO" << '\n';
            return;
        }
    }
    ops.clear();
    for (int i = 1; i <= n; i++) {
        int prev = i;                    // 刚顶上来的格子，一开始是第 i 格
        for (int v = a[i]; v <= i - 1; v++) {
            int x = cur[v];              // 原本拿着 v 的格子，要把它推到 v+1
            ops.push_back(cur[x]);       // 找一个值为 x 的格子当操作下标
            cur[v] = prev;               // v 现在由 prev 拿着
            prev = x;
        }
        cur[i] = prev;
    }
    cout << "YES" << '\n' << ops.size() << '\n';
    for (int x : ops)
        cout << x << ' ';
    cout << '\n';
}
signed main () {
    ios::sync_with_stdio (false);
    cin.tie (0);
    cout.tie (0);
    cin >> T;
    while (T--)
        solve ();
    return 0;
}

第一步　先想两个最简单的情况（各 10 分，各一行代码）

第二步　没有“距离豁免”时：树形 DP
以 k 为根，从叶子往上算。一个点 v 能不能共振，要看父亲或某个儿子是不是核。父亲在 v 的子树外面，算子树时不知道，所以分两种情况假设：父亲是核 / 父亲不是核。再加上 v 自己当核，一共三个数。

core(v) = Σ g1(儿子)　　v 当核拿 0，每个儿子都有了“父亲是核”的待遇

g1(v) = max( core(v), max(m_v, w_v) + Σ g0(儿子) )　　父亲是核，v 想共振就共振

g0(v) = max( core(v), m_v + Σ g0(儿子), w_v + Σ g0(儿子) + max_c (core(c) − g0(c)) )

signed main () {
    ios::sync_with_stdio (false);
    cin.tie (0);
    cout.tie (0);
    cin >> n >> q;
    for (int i = 1; i <= n; i++)
        cin >> m[i];
    for (int i = 1; i <= n; i++)
        cin >> w[i];
    cin >> k;
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        e[u].push_back(v);
        e[v].push_back(u);
    }
    vector <int> ord;                    // BFS 序，倒着扫就是从叶子往上
    ord.push_back(k);
    for (int h = 0; h < (int)ord.size(); h++) {
        int u = ord[h];
        for (int v : e[u]) {
            if (v == fa[u])
                continue;
            fa[v] = u;
            dep[v] = dep[u] + 1;
            ord.push_back(v);
        }
    }
    for (int h = n - 1; h >= 0; h--) {
        int u = ord[h];
        int s0 = 0, s1 = 0, pen = LLONG_MIN;
        for (int v : e[u]) {
            if (v == fa[u])
                continue;
            s0 += g0[v];
            s1 += g1[v];
            pen = max (pen, cr[v] - g0[v]);
        }
        cr[u] = s1;
        g0[u] = max (cr[u], m[u] + s0);
        if (pen != LLONG_MIN)            // 有儿子才能靠儿子当核来共振
            g0[u] = max (g0[u], w[u] + s0 + pen);
        g1[u] = max (cr[u], max (m[u], w[u]) + s0);
    }
    int md = 0;
    for (int i = 1; i <= n; i++) {
        md = max (md, dep[i]);
        sum[dep[i]] += max (m[i], w[i]);
        lay[dep[i]] += g1[i];
    }
    for (int d = 1; d <= md; d++)
        sum[d] += sum[d - 1];
    while (q--) {
        int L;
        cin >> L;
        if (L >= md)
            cout << sum[md] << '\n';
        else
            cout << (L > 0 ? sum[L - 1] : 0) + lay[L] << '\n';
    }
    return 0;
}

            for (int r = 0; r < a; r++)
                dis[r] = LLONG_MAX;
            dis[0] = 0;
            priority_queue <pair <int, int>, vector <pair <int, int> >, greater <pair <int, int> > > pq;
            pq.push({0, 0});
            while (!pq.empty()) {
                int d = pq.top().first, u = pq.top().second;
                pq.pop();
                if (d != dis[u])
                    continue;
                for (int j = 1; j < (int)p.size(); j++) {
                    int v = (u + p[j]) % a;
                    if (d + p[j] < dis[v]) {
                        dis[v] = d + p[j];
                        pq.push({dis[v], v});
                    }
                }
            }
            for (int i : id)
                ans[i] = (dis[n[i] % a] <= n[i]);
        }
    }
    for (int i = 1; i <= t; i++)
        cout << (ans[i] ? "YES" : "NO") << '\n';
    return 0;
}
